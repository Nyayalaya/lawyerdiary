📊 CODE CHANGES MADE
═════════════════════════════════════════════════════════════════════════════

BEFORE:
  builder.Services.AddOpenApi();                    // Only OpenAPI
  
  if (app.Environment.IsDevelopment())
  {
      app.MapOpenApi();                             // Only spec endpoint
  }

AFTER:
  builder.Services.AddSwaggerGen(options =>         // ✅ NEW
  {                                                 // ✅ NEW
      options.SwaggerDoc("v1", ...);               // ✅ NEW
      options.AddSecurityDefinition("Bearer", ...); // ✅ NEW
      options.IncludeXmlComments(xmlPath);         // ✅ NEW
  });                                               // ✅ NEW
  
  if (app.Environment.IsDevelopment())
  {
      app.UseSwagger();                             // ✅ NEW
      app.UseSwaggerUI(c =>                         // ✅ NEW
      {                                             // ✅ NEW
          c.SwaggerEndpoint("/swagger/v1/swagger.json", "CourtApp.Api v1");
          c.RoutePrefix = "";                       // Serve at root
      });                                           // ✅ NEW
      app.MapOpenApi();
  }
